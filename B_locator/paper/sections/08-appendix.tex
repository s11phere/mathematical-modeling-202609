\section{问题二主要源程序}\label{app:q2code}
问题二的完整源程序（支撑材料 \texttt{Q2/src/q2\_solution.py}）见下，仅保留与正文

\begin{lstlisting}[language=python, caption={问题二：期望定位区域直径与最优点、候选区域（Python 3）}]
import math
import numpy as np

X_MAX, ALPHA_DEG = 1500.0, 1.0        # 有效接收半径；示向度误差界
ALPHA = math.radians(ALPHA_DEG)       # 探测束半张角


def quad_vertices(x_t, a, b, alpha=ALPHA):
    """S1 的两条边界射线（与 x 轴成 +-alpha）与 S2 的两条边界射线
    （绕 S2->S 方向成 +-alpha）两两求交，得交会四边形 ABCD 的四个顶点。"""
    s2 = np.array([a, b])
    psi = math.atan2(-b, x_t - a)      # S 相对 S2 的方位角
    pts = []
    for t1 in (alpha, -alpha):
        u1 = np.array([math.cos(t1), math.sin(t1)])
        for t2 in (psi + alpha, psi - alpha):
            u2 = np.array([math.cos(t2), math.sin(t2)])
            den = u1[0] * u2[1] - u1[1] * u2[0]
            if abs(den) < 1e-14:       # 两射线平行：定位区域无界
                return None
            s = (s2[0] * u2[1] - s2[1] * u2[0]) / den
            pts.append(s * u1)
    return np.array(pts)


def quad_diameter(x_t, a, b, alpha=ALPHA):
    """定位区域直径 D = 四边形顶点两两距离的最大值。"""
    p = quad_vertices(x_t, a, b, alpha)
    if p is None:
        return math.inf
    d = p[:, None, :] - p[None, :, :]
    return float(np.sqrt((d ** 2).sum(-1)).max())


def dbar(a, b, n=800, alpha=ALPHA):
    """期望定位区域直径：按 P_T(x_T|S1) = 2 x_T / x_max^2 加权求和。"""
    if abs(b) < 1e-9:                  # 两点与源点共线，区域无界
        return math.inf
    x_t = X_MAX * (np.arange(n) + 0.5) / n
    w = 2.0 * x_t / X_MAX ** 2
    d = np.array([quad_diameter(float(x), a, b, alpha) for x in x_t])
    if not np.isfinite(d).all():
        return math.inf
    return float((w * d).sum() / w.sum())


if __name__ == "__main__":             # 自检：粗网格上的最优相对位形
    best = min(((dbar(x * 1.0, y * 1.0), x, y)
                for x in range(950, 1251, 25) for y in range(500, 901, 25)),
               key=lambda t: t[0])
    print("Dbar_min = %.2f m at (a,b) = (%.0f, %.0f) m" % best)
\end{lstlisting}
