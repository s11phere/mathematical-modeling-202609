\section{问题一主要源程序}\label{app:code}
问题一的完整源程序（支撑材料 \texttt{code/p1\_locator.py}）见下；其中凸包采用 Andrew
单调链，仅列出与正文式~\eqref{eq:wedge}--\eqref{eq:cover} 直接对应的部分。

\begin{lstlisting}[language=python, caption={问题一：交会定位区域直径与直径圆覆盖判定（Python 3）}]
import math
EPS, BOX = 1.0, 1.0e7          # 示向度误差界；裁剪初始包围盒


def rot(v, deg):
    a = math.radians(deg)
    return (v[0] * math.cos(a) - v[1] * math.sin(a),
            v[0] * math.sin(a) + v[1] * math.cos(a))


def dist(p, q):
    return math.hypot(p[0] - q[0], p[1] - q[1])


def bounded(svds, eps=EPS):
    """角楔交集有界 <=> 有向方位角区间公共交集为空。"""
    segs = [(max(0.0, (svds[0] - eps) % 360.0), min(360.0, (svds[0] + eps) % 360.0))]
    for th in svds[1:]:
        lo, hi = (th - eps) % 360.0, (th + eps) % 360.0
        parts = [(lo, hi)] if lo <= hi else [(lo, 360.0), (0.0, hi)]
        new = []
        for a, b in segs:
            for c, d in parts:
                if max(a, c) <= min(b, d) + 1e-12:
                    new.append((max(a, c), min(b, d)))
        segs = new
        if not segs:
            return True
    return False


def clip(poly, n, b):
    """保留半平面 {x : n.x + b >= 0}。"""
    out, k = [], len(poly)
    for i in range(k):
        p, q = poly[i], poly[(i + 1) % k]
        fp = n[0] * p[0] + n[1] * p[1] + b
        fq = n[0] * q[0] + n[1] * q[1] + b
        if fp >= 0.0:
            out.append(p)
        if (fp < 0.0 < fq) or (fp > 0.0 > fq):
            t = fp / (fp - fq)
            out.append((p[0] + t * (q[0] - p[0]), p[1] + t * (q[1] - p[1])))
    return out


def region(dets, eps=EPS):
    """定位区域 = 全部角楔之交。dets = [(x, y, theta), ...]"""
    poly = [(-BOX, -BOX), (BOX, -BOX), (BOX, BOX), (-BOX, BOX)]
    for (x, y, th) in dets:
        for s in (-1.0, 1.0):
            a = math.radians(th + s * eps)
            r = (math.cos(a), math.sin(a))        # 边界射线方向
            n = rot(r, -90.0 * s)                 # 指向楔内的法向量
            poly = clip(poly, n, -(n[0] * x + n[1] * y))
            if not poly:
                return []
    return poly


def cross(o, a, b):
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def hull(pts):
    """Andrew 单调链（逆时针，弹出共线中间点）。"""
    pts = sorted(set(pts))
    if len(pts) <= 2:
        return pts
    lo, hi = [], []
    for p in pts:
        while len(lo) >= 2 and cross(lo[-2], lo[-1], p) <= 0:
            lo.pop()
        lo.append(p)
    for p in reversed(pts):
        while len(hi) >= 2 and cross(hi[-2], hi[-1], p) <= 0:
            hi.pop()
        hi.append(p)
    return lo[:-1] + hi[:-1]


def solve(dets, eps=EPS):
    """返回直径、直径端点、圆心与覆盖结论。"""
    if not bounded([d[2] for d in dets], eps):
        return {"status": "unbounded"}
    V = hull(region(dets, eps))
    if len(V) < 3:
        return {"status": "degenerate", "vertices": V}
    best, (A, B) = -1.0, (V[0], V[0])
    for i in range(len(V)):
        for j in range(i + 1, len(V)):
            if dist(V[i], V[j]) > best:
                best, (A, B) = dist(V[i], V[j]), (V[i], V[j])
    M = ((A[0] + B[0]) / 2.0, (A[1] + B[1]) / 2.0)      # 直径圆圆心
    mx = max(dist(M, p) for p in V)
    scale = max(1.0, max(max(abs(x), abs(y)) for (x, y, _) in dets))
    return {"status": "bounded", "vertices": V, "diameter": best,
            "endpoints": (A, B), "center": M, "radius": best / 2.0,
            "max_vertex_dist": mx,
            "coverage": mx <= best / 2.0 + 1e-9 * scale}


if __name__ == "__main__":                      # 以算例 A1 自检
    dets = [(-432.855, -596.368, 91.86), (-290.614, -501.995, 121.82),
            (474.860, -160.806, 186.02), (461.755, 349.174, 212.80),
            (-444.749, -62.142, 269.79), (-958.142, 736.087, 298.12)]
    r = solve(dets)
    print("D = %.4f m, max|P-M| = %.4f m, covers = %s"
          % (r["diameter"], r["max_vertex_dist"], r["coverage"]))
\end{lstlisting}
