  {-1.5ex\@plus -1ex \@minus -.2ex}%
  {0.5ex \@plus .2ex}%
  {\normalfont\heiti\bfseries\normalsize}}
% 正文、\songti 与模板 \song 共用自带宋体；所有 CJK 家族的粗体均用
% 同一常规黑体。第二个参数提供黑体文件的路径/字库索引。
\newcommand*\mcm@setupCJKfonts[2]{%
  \setCJKmainfont[Path=fonts/,BoldFont={#1},BoldFeatures={#2},BoldItalicFont={#1},BoldItalicFeatures={#2}]{simsun.ttc}%
  \setCJKfamilyfont{zhsong}[Path=fonts/,BoldFont={#1},BoldFeatures={#2},BoldItalicFont={#1},BoldItalicFeatures={#2}]{simsun.ttc}%
  \setCJKfamilyfont{song}[Path=fonts/,BoldFont={#1},BoldFeatures={#2},BoldItalicFont={#1},BoldItalicFeatures={#2}]{simsun.ttc}%
  \setCJKsansfont[#2,BoldFont={#1},BoldFeatures={#2},BoldItalicFont={#1},BoldItalicFeatures={#2}]{#1}%
  \setCJKfamilyfont{zhhei}[#2,BoldFont={#1},BoldFeatures={#2},BoldItalicFont={#1},BoldItalicFeatures={#2}]{#1}%
  \setCJKmonofont[Path=fonts/,BoldFont={#1},BoldFeatures={#2},BoldItalicFont={#1},BoldItalicFeatures={#2}]{simsun.ttc}%
  \setCJKfamilyfont{zhkai}[Path=fonts/,BoldFont={#1},BoldFeatures={#2},BoldItalicFont={#1},BoldItalicFeatures={#2}]{simkai.ttf}%
  \setCJKfamilyfont{kai}[Path=fonts/,BoldFont={#1},BoldFeatures={#2},BoldItalicFont={#1},BoldItalicFeatures={#2}]{simkai.ttf}%
}

% 参考文献：收紧条目间距（thebibliography 用原生 \list，不受上面 \setlist 影响）
\renewenvironment{thebibliography}[1]
     {\section*{\refname}%
      \@mkboth{\MakeUppercase\refname}{\MakeUppercase\refname}%
      \list{\@biblabel{\@arabic\c@enumiv}}%
           {\settowidth\labelwidth{\@biblabel{#1}}%
            \leftmargin\labelwidth
            \advance\leftmargin\labelsep
            \itemsep=0pt\parsep=0pt\topsep=0pt
            \@openbib@code
            \usecounter{enumiv}%
            \let\p@enumiv\@empty
            \renewcommand\theenumiv{\@arabic\c@enumiv}}%
      \sloppy
      \clubpenalty4000
      \@clubpenalty \clubpenalty
      \widowpenalty4000%
      \sfcode`\.\@m}
     {\def\@noitemerr
       {\@latex@warning{Empty `thebibliography' environment}}%
      \endlist}
\makeatother
